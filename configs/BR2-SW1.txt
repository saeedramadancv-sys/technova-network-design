! =====================================================================
!  DEVICE : BR2-SW1           MODEL : Cisco 2960-24TT

!  ROLE   : Branch access layer
!  UPLINK : Fa0/1 -> BR2-R1 Gi0/0 (802.1Q trunk, router-on-a-stick)

enable
configure terminal
!
! ---------------------------------------------------------------------
!  1. IDENTITY AND BASELINE
! ---------------------------------------------------------------------
hostname BR2-SW1
no ip domain-lookup
ip domain-name technova.local
!
enable secret NetW0rks#2026
service password-encryption
username netadmin privilege 15 secret Adm1n#Tech2026
!
service timestamps log datetime msec localtime
clock timezone AST 3
login block-for 120 attempts 4 within 60
!
banner motd #
 *** TechNova Solutions - BR2 ACCESS SWITCH - authorised use only ***
#
!
! ---------------------------------------------------------------------
!  2. MANAGEMENT ACCESS
! ---------------------------------------------------------------------
crypto key generate rsa general-keys modulus 1024
ip ssh version 2
!
ip access-list standard MGMT-ACCESS
 permit 172.16.1.48 0.0.0.15
 deny   any log
exit
!
line console 0
 login local
 exec-timeout 15 0
 logging synchronous
exit
!
line vty 0 15
 access-class MGMT-ACCESS in
 transport input ssh
 login local
 exec-timeout 10 0
 logging synchronous
exit
!
! ---------------------------------------------------------------------
!  3. VLANS
! ---------------------------------------------------------------------
vlan 210
 name BR2-STAFF
exit
vlan 240
 name BR2-GUEST
exit
vlan 299
 name BR2-MGMT
exit
vlan 999
 name PARKING-NATIVE
exit
!
spanning-tree mode rapid-pvst
spanning-tree vlan 210,240,299 root primary
!
! ---------------------------------------------------------------------
!  4. UPLINK TRUNK TO THE ROUTER
! ---------------------------------------------------------------------
interface FastEthernet0/1
 description ### TRUNK TO BR2-R1 Gi0/0 ###
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 210,240,299
 switchport nonegotiate
 ip dhcp snooping trust
 no shutdown
exit
!
! ---------------------------------------------------------------------
!  5. STAFF ACCESS PORTS (VLAN 210)
! ---------------------------------------------------------------------
interface range FastEthernet0/5-12
 description ### BR2 STAFF WORKSTATIONS ###
 switchport mode access
 switchport access vlan 210
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
exit
!
! ---------------------------------------------------------------------
!  6. GUEST WIRELESS PORT (VLAN 240)
! ---------------------------------------------------------------------
interface FastEthernet0/20
 description ### BR2 GUEST WIRELESS AP ###
 switchport mode access
 switchport access vlan 240
 switchport protected
 switchport port-security
 switchport port-security maximum 6
 switchport port-security violation restrict
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
exit
!
! ---------------------------------------------------------------------
!  7. UNUSED PORTS
! ---------------------------------------------------------------------
interface range FastEthernet0/2-4
 description ### UNUSED ###
 switchport mode access

 shutdown
exit

 description ### UNUSED ###
 switchport mode access
 switchport access vlan 999
 shutdown
exit
!
interface range FastEthernet0/21-24
 description ### UNUSED ###
 switchport mode access
 switchport access vlan 999
 shutdown
exit
!
interface range GigabitEthernet0/1-2
 description ### UNUSED UPLINK PORTS ###
 shutdown
exit
!

!  8. DHCP SNOOPING
! ---------------------------------------------------------------------
ip dhcp snooping
ip dhcp snooping vlan 210,240,299
no ip dhcp snooping information option
!
! ---------------------------------------------------------------------
!  9. MANAGEMENT INTERFACE
! ---------------------------------------------------------------------
interface Vlan299
 description ### SWITCH MANAGEMENT ADDRESS ###
 ip address 172.16.20.26 255.255.255.248
 no shutdown
exit
!

!

ntp server 172.16.255.241
